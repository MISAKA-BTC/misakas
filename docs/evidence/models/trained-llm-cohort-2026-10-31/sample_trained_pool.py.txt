import concurrent.futures,json,urllib.request,random,hashlib
from pathlib import Path
root=Path(__file__).parent
repos=['openai-community/gpt2','distilbert/distilgpt2','EleutherAI/pythia-70m','EleutherAI/pythia-160m','HuggingFaceTB/SmolLM-135M','HuggingFaceTB/SmolLM2-135M','HuggingFaceTB/SmolLM2-360M']
def read(repo):
 info=json.load(urllib.request.urlopen(f'https://huggingface.co/api/models/{repo}?blobs=true',timeout=60));rev=info['sha'];base=f'https://huggingface.co/{repo}/resolve/{rev}/'
 config=urllib.request.urlopen(base+'config.json',timeout=60).read();card=urllib.request.urlopen(base+'README.md',timeout=60).read();p=root/'pool'/repo.replace('/','__');p.mkdir(parents=True);(p/'info.json').write_text(json.dumps(info,indent=2));(p/'config.json').write_bytes(config);(p/'README.md.txt').write_bytes(card)
 return {'repo':repo,'revision':rev,'config':json.loads(config),'card_sha256':hashlib.sha256(card).hexdigest(),'training_evidence':f'https://huggingface.co/{repo}/blob/{rev}/README.md','safetensors':info.get('safetensors')}
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool: members=list(pool.map(read,repos))
seed=random.SystemRandom().getrandbits(64);selected=random.Random(seed).sample(members,3)
record={'scope':'uniform sample of three from seven explicitly predeclared publisher-released pretrained causal LMs; not a Hub-wide random sample','exclusions':'random/untrained/testing repos excluded; source/checkpoint <= 1.5 GiB; sequential/parallel execution does not alter membership','pool':members,'seed':seed,'selected':selected,'resampling_on_failure':False}
(root/'selection.json').write_text(json.dumps(record,indent=2)+'\n');print('seed',seed)
for m in selected:print(m['repo'],m['revision'],m['config'].get('max_position_embeddings',m['config'].get('n_positions')))
